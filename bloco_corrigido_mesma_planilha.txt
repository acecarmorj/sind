function _getSpreadsheet_() {
  var id = String(SIND_SETUP.SPREADSHEET_ID || '').trim();

  if (!id) {
    throw new Error('SPREADSHEET_ID não configurado.');
  }

  return SpreadsheetApp.openById(id);
}
